// XXX: 性能问题 - O(n²) 复杂度
export function slowFunction(arr: number[]) {
  const results: number[] = [];
  // TODO: 优化算法，使用更高效的方法
  for (let i = 0; i < arr.length; i++) {
    for (let j = 0; j < arr.length; j++) {
      results.push(arr[i] + arr[j]);
    }
  }
  return results;
}
